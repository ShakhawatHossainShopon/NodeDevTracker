Project Lab 05 Chunked HTTP

ল্যাব ০৫: লাইভ চাঙ্কড HTTP স্ট্রিমিং সার্ভার

সার্ভার-সেন্ট-ইভেন্টস (SSE) ও চাঙ্কড রেসপন্স ইঞ্জিন।

🛠️ কোড বাস্তবায়ন

import http from 'node:http';

const server = http.createServer((req, res) => {
  if (req.url === '/events') {
    res.writeHead(200, {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      'Connection': 'keep-alive'
    });

    let count = 0;
    const interval = setInterval(() => {
      count++;
      const payload = JSON.stringify({ tick: count, timestamp: new Date().toISOString() });
      res.write('data: ' + payload + '\\n\\n');

      if (count >= 3) {
        clearInterval(interval);
        res.end();
      }
    }, 1000);
  }
});

server.listen(3000, () => console.log('SSE Streamer listening on port 3000'));
