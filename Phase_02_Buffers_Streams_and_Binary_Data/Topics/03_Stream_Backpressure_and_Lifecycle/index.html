Stream Backpressure ও ইভেন্ট লাইফসাইকেল (data, drain, finish)

উচ্চগতির ডাটা সোর্স এবং ধীরগতির গন্তব্যের মধ্যে মেমোরি ব্যালেন্স ও ব্যাকপ্রেশার ইঞ্জিনিয়ারিং।

১. ব্যাকপ্রেশার (Backpressure) কী?

যখন কোনো রিডেবল স্ট্রিম অতি দ্রুত গতিতে ডাটা উৎপাদন করে, কিন্তু রাইটেবল স্ট্রিম ধীরগতির কারণে সেই গতিতে ডিস্কে বা ইন্টারনেটে লিখতে পারে না, তখন অতিরিক্ত ডাটা মেমোরির ইন্টারনাল বাফারে জমতে শুরু করে। এই অবস্থাকে Backpressure বলে।

⚠️ মারাত্মক ঝুঁকি: ব্যাকপ্রেশার হ্যান্ডেল না করলে Writable স্ট্রিমের বাফার সীমাহীন বড় হয়ে সার্ভারের সমস্ত র‍্যাম (RAM) গ্রাস করবে এবং প্রোডাকশন ক্র্যাশ ঘটবে!

২. highWaterMark এবং drain ইভেন্ট

প্রতিটি Writable স্ট্রিমে একটি সর্বোচ্চ মেমোরি সীমা থাকে যাকে highWaterMark বলা হয় (ডিফল্ট ১৬ কিলোবাইট)।

  • writable.write(chunk) মেথড false রিটার্ন করলে বুঝতে হবে ইন্টারনাল বাফার পূর্ণ!
  • বাফার খালি হওয়া পর্যন্ত রিডেবল স্ট্রিমকে পজ (pause) করতে হবে।
  • বাফার খালি হলে রাইটেবল স্ট্রিম drain ইভেন্ট ফায়ার করে। তখন পুনরায় ডাটা পাঠানো শুরু করা যায়।
import fs from 'node:fs';

function writeSafely(writer, totalRecords) {
  let count = 0;

  function write() {
    let canContinue = true;
    while (count < totalRecords && canContinue) {
      count++;
      const data = 'Log entry #' + count + '\\n';
      if (count === totalRecords) {
        writer.end(data);
      } else {
        canContinue = writer.write(data);
      }
    }

    if (count < totalRecords) {
      // বাফার পূর্ণ! drain ইভেন্টের অপেক্ষা করুন
      writer.once('drain', write);
    }
  }

  write();
}

const fileStream = fs.createWriteStream('safe_output.txt');
writeSafely(fileStream, 100000);
