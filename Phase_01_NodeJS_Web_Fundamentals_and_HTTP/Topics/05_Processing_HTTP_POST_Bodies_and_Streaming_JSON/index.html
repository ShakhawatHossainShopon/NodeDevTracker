Processing HTTP POST/PUT Bodies & Streaming JSON Payloads

নেটিভ Node.js-এ স্ট্রিম আকারে ইনকামিং ডাটা সংগ্রহ করা, বাফার জোড়া লাগানো, নিরাপদ JSON পার্সিং এবং সঠিক HTTP স্ট্যাটাস কোড প্রদান।

১. কেন Node.js-এ `req.body` সরাসরি থাকে না?

Express.js বা অন্যান্য ফ্রেমোর্ক ব্যবহার করলে আমরা সহজেই req.body পেয়ে যাই। কিন্তু নেটিভ Node.js-এ req.body বলে কোনো প্রোপার্টি আগে থেকে থাকে না!

কারণ: HTTP রিকোয়েস্ট অবজেক্ট req হলো একটি Readable Stream। যখন ক্লায়েন্ট কোনো বড় JSON বা ফাইল আপলোড করে, সেই ডাটা একবারে না এসে ইন্টারনেটের ভেতর দিয়ে ছোট ছোট বাইনারি বাফারের খণ্ডে (Chunks) সার্ভারে পৌঁছায়।

💡 স্ট্রিমিং আর্কিটেকচারের সুবিধা:
Node.js যদি সম্পূর্ণ ডাটা আসা পর্যন্ত মেমোরি ব্লক করে রাখত, তবে একসাথে লাখ লাখ রিকোয়েস্ট আসলে সার্ভারের র‍্যাম (RAM) ক্র্যাশ করত। স্ট্রিমিংয়ের কারণে ডাটা আসার সাথে সাথেই প্রসেস বা ডিস্কে পাইপ করা যায়।

২. চাঙ্ক (Chunk) সংগ্রহ ও বাফার মার্জ করার পদ্ধতি

ইনকামিং রিকোয়েস্ট থেকে বডি পাওয়ার মূল নিয়ম হলো দুটি ইভেন্ট শোনা:

  1. req.on('data', chunk => { ... }): নেটওয়ার্ক থেকে প্রতিটি ডাটা খণ্ড আসার সময় এটি চলে।
  2. req.on('end', () => { ... }): ক্লায়েন্ট যখন সব ডাটা পাঠিয়ে শেষ করে তখন এটি ট্রিগার হয়।
import http from 'node:http';

const server = http.createServer((req, res) => {
  if (req.method === 'POST' && req.url === '/api/users') {
    const chunks = [];

    // প্রতিবার ডাটা চাঙ্ক আসলে অ্যারেতে রাখা
    req.on('data', (chunk) => {
      chunks.push(chunk);
    });

    // ডাটা পাঠানো শেষ হলে সব বাফার একত্র করা
    req.on('end', () => {
      const rawBody = Buffer.concat(chunks).toString('utf-8');
      console.log(`[RAW BODY RECEIVED] Length: ${rawBody.length} chars`);
    });
  }
});

৩. নিরাপদ JSON পার্সিং ও ভ্যালিডেশন

ক্লায়েন্ট সবসময় সঠিক JSON পাঠাবে—এর কোনো নিশ্চয়তা নেই। ক্লায়েন্ট ভুল সিনট্যাক্স পাঠালে সরাসরি JSON.parse() কল করলে নোড প্রসেস সঙ্গে সঙ্গে ক্র্যাশ করবে!

⚠️ ক্র্যাশ এড়াতে সর্বদা `try ... catch` ব্যবহার করুন:
ভুল JSON আসলে 400 Bad Request পাঠাতে হবে, সার্ভার বন্ধ হওয়া যাবে না। 
req.on('end', () => {
  try {
    const rawBody = Buffer.concat(chunks).toString('utf-8');
    const body = JSON.parse(rawBody || '{}');

    // ফিল্ড ভ্যালিডেশন
    if (!body.name || !body.email) {
      res.statusCode = 400; // Bad Request
      return res.end(JSON.stringify({ error: 'Name and email are required' }));
    }

    // নতুন রিসোর্স সফলভাবে তৈরি হলে ২০১ স্ট্যাটাস কোড
    res.statusCode = 201; // Created
    res.end(JSON.stringify({ status: 'success', data: body }));
  } catch (err) {
    res.statusCode = 400;
    res.end(JSON.stringify({ error: 'Invalid JSON format' }));
  }
});
