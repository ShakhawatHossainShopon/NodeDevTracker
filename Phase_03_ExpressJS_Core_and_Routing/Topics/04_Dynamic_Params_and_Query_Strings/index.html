Handling Dynamic Route Parameters (req.params) & Query Strings (req.query)

ডায়নামিক URL রাউটিং, নেস্টেড প্যারামিটার ও কুয়েরি ফিল্টারিং/প্যাজিনেশন হ্যান্ডলিং।

১. Route Parameters (`req.params`)

রুট প্যাথে কোলন (:) চিহ্ন দিয়ে প্যারামিটার নির্ধারণ করা হয়। Express স্বয়ংক্রিয়ভাবে এটি পার্স করে req.params অবজেক্টে প্রোভাইড করে:

import express from 'express';

const app = express();

// নেস্টেড ডায়নামিক রুট
app.get('/users/:userId/orders/:orderId', (req, res) => {
  const { userId, orderId } = req.params;
  res.json({
    userFound: userId,
    orderFound: orderId,
    timestamp: new Date()
  });
});

app.listen(3000);

২. Query Strings (`req.query`) — ফিল্টারিং ও প্যাজিনেশন

URL কুয়েরি প্যারামিটারগুলো (যেমন ?category=books&page=2&limit=10) স্বয়ংক্রিয়ভাবে req.query তে পার্সড অবজেক্ট হিসেবে থাকে:

app.get('/api/products', (req, res) => {
  const category = req.query.category || 'all';
  const page = parseInt(req.query.page) || 1;
  const limit = parseInt(req.query.limit) || 20;

  res.json({
    filter: category,
    pagination: { page, limit },
    results: []
  });
});
