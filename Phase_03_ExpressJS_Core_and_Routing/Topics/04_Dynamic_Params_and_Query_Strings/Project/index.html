Project Lab 04 Params & Query

ল্যাব ০৪: মাল্টি-প্যারামিটার সার্চ ও ফিল্টারিং সার্ভিস

req.params এবং req.query এর ইন্টিগ্রেটেড হ্যান্ডলিং ল্যাব।

🛠️ কোড বাস্তবায়ন

import express from 'express';

const app = express();

const inventory = [
  { id: 'p101', category: 'phones', name: 'iPhone 15 Pro', price: 999 },
  { id: 'p102', category: 'phones', name: 'Pixel 8', price: 699 },
  { id: 'p103', category: 'laptops', name: 'MacBook Air M3', price: 1099 }
];

app.get('/catalog/:category/:productId', (req, res) => {
  const { category, productId } = req.params;
  const currency = req.query.currency || 'USD';

  const product = inventory.find(p => p.id === productId && p.category === category);
  if (!product) {
    return res.status(404).json({ error: 'Product not found in this category' });
  }

  res.json({
    product,
    pricing: { currency, formattedPrice: currency + ' ' + product.price }
  });
});

app.listen(3000, () => console.log('Catalog service ready on port 3000'));
