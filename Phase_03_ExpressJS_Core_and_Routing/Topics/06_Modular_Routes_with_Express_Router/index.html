Modular Route Organization using express.Router()

মনোলিথিক কোডবেস ভেঙে মডুলার, রিসোর্স-ভিত্তিক রাউটার আর্কিটেকচার।

১. কেন `express.Router()` প্রয়োজন?

একটি মাঝারি বা বড় অ্যাপ্লিকেশনে যদি সব রুট একটিমাত্র server.js ফাইলে রাখা হয়, তবে কয়েক হাজার লাইনের কোড তৈরি হয় যা মেইনটেইন করা দুঃসাধ্য। express.Router() একটি মিনি-অ্যাপ্লিকেশন যা নিজস্ব মিডলওয়্যার ও রুট ধারণ করতে পারে এবং মূল অ্যাপ্লিকেশনের নির্দিষ্ট প্রিফিক্সে মাউন্ট করা যায়।

💡 আর্কিটেকচারাল প্যাটার্ন: ইউজার সম্পর্কিত রুট routes/userRoutes.js, প্রোডাক্ট সম্পর্কিত রুট routes/productRoutes.js ফাইলে আলাদা রেখে app.use('/api/v1/users', userRouter) দিয়ে যুক্ত করা হয়।

২. কোড স্ট্রাকচার: মডুলার রাউটার উদাহরণ

// routes/userRoutes.js
import { Router } from 'express';

const router = Router();

router.get('/', (req, res) => {
  res.json({ users: [{ id: 1, name: 'Shopon' }] });
});

router.get('/:id', (req, res) => {
  res.json({ user: { id: req.params.id, name: 'User Profile' } });
});

export default router;

// app.js
import express from 'express';
import userRouter from './routes/userRoutes.js';

const app = express();
app.use('/api/v1/users', userRouter);

app.listen(3000);
